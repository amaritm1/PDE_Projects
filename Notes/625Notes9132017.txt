#<stdio.h>
#<stdlib.h>


void *malloc_or_exit(size_t nbytes)
{
	void *x=malloc(nbytes); //Grabs 80 bytes of the memory and returns a pointer to this memory and completes the defintion of x where nbytes is the number of bytes
	if(x==NULL)
	{
		fprintf("stderr,"malloc failed\n");
		exit(EXIT_FAILURE);//Kills program
	}
	
	return x;

}


int main(void)

{
	double *x;
	int n=10;
	x=malloc_or_exit(n * sizeof *x); 
	if (x==NULL)
	{
	
		fprintf(stderr,"malloc failed on line %d in file %s\n",__LINE__,__FILE__); //there are two underscores all four times
		return EXIT_FAILURE;
	}	
	for(int i=0;i<n;i++)

	{
		x[i]=1.0/(i+1);
		printf("x[%d]=%g\n",i,x[i]);

	}
	
	free(x); // frees up the memeory of x every malloc should have a matching free
	return EXIT_SUCCESS; //same as return 0	These return values go to the shell and can be checked by typing the command "echo $status"	
}
