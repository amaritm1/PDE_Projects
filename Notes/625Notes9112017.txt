	int main(int argc, char **argv) //FIRST VERSION IGNORES COMMAND LINE ARGUMENTS THE SECOND CAPTURES THEM
	{
		
		printf("program invoked with %d arguments\n", argc);
		for(int i=0; i<argc;i++)
		{
			printf("arg %d is &s\n",i, argv[i]);
		}

	}
