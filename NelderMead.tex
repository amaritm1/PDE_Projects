	\begin{definition}
		\textbf{Simplex} Given n linearly independent vectors connect each of the end points to get n+1 vertecies.
	\end{definition}

Given a simplex in $ \RRn $ let $ \ver{x}{0},\ver{x}{1},\dots,\ver{x}{n} $ denote the verticies. Let $ y_i=f(\ver{x}{i}) $ for $ i=1\dots n.$ Denote $ y_{iz} $ as the farthest away from the minimum, $ y_{iy} $ as the second farthest, and $ y_{ia}$ is the closest to the minimum. Notice the indicies may not be uniquely determined, but we want $ ia\neq iy,\; iy\neq iz,\text{ and } iz \neq ia. $ We call $ y_{ia}=\min_i y_i $, $y_{iz}=\max_{i} y_i$, $y_{iy}=\max_{i \neq z} y_i.$  

\begin{definition}
	\text{centroid} denoted as $ \hat{x}=\frac{1}{n}\sum_{i=0}^{n} \ver{x}{i} $ for $ i\neq iz $.
\end{definition}

We denote $ \ver{x}{r} $ as the the reflextion and $ \ver{x}{e} $ as the extension.
